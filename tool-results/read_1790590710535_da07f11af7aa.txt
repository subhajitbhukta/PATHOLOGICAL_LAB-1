    13→const inv = (
    14→  id: string, date: string, scope: Invoice["scope"], billToId: string, billTo: string,
    15→  billToSub: string | undefined, gstin: string, lines: [string, number, number][],
    16→  discount: number, paid: number, mode: string, status: PayStatus, dueDate?: string, orderId?: string,
    17→): Invoice => {
    18→  const subtotal = lines.reduce((a, [, q, r]) => a + q * r, 0);
    19→  const taxable = subtotal - discount;
    20→  const cgst = Math.round(taxable * 0.09);
    21→  return {
    22→    id, date, dueDate, scope, billToId, billTo, billToSub, gstin,
    23→    lines: lines.map(([d, q, r]) => ({ description: d, hsn: "999311", qty: q, rate: r })),
    24→    subtotal, discount, taxable, cgst, sgst: cgst, total: taxable + cgst * 2,
    25→    paid, due: taxable + cgst * 2 - paid, mode, status, orderId,
    26→  };
    27→};
