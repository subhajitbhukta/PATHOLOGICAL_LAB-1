   257→   257→export interface Column<T> {
   258→   258→  key: string;
   259→   259→  header: string;
   260→   260→  className?: string;
   261→   261→  headClassName?: string;
   262→   262→  render?: (row: T) => React.ReactNode;
   263→   263→  value?: (row: T) => string | number;
   264→   264→}
   265→   265→
   266→   266→export function DataTable<T extends object>({
   267→   267→  columns, rows, searchOf, searchPlaceholder = "Search…", pageSize = 10, onRowClick,
   268→   268→  toolbar, filters, dense,
   269→   269→}: {
   270→   270→  columns: Column<T>[]; rows: T[]; searchOf?: (row: T) => string;
   271→   271→  searchPlaceholder?: string; pageSize?: number; onRowClick?: (row: T) => void;
   272→   272→  toolbar?: React.ReactNode; filters?: React.ReactNode; dense?: boolean;
   273→   273→}) {
   274→   274→  const [q, setQ] = React.useState("");
   275→   275→  const [page, setPage] = React.useState(0);
   276→   276→
   277→   277→  React.useEffect(() => setPage(0), [q, rows.length]);
   278→   278→
   279→   279→  const filtered = React.useMemo(() => {
   280→   280→    if (!q.trim()) return rows;
   281→   281→    const needle = q.toLowerCase();
   282→   282→    return rows.filter((r) =>
   283→   283→      searchOf ? searchOf(r).toLowerCase().includes(needle) : JSON.stringify(r).toLowerCase().includes(needle),
   284→   284→    );
   285→   285→  }, [rows, q, searchOf]);
   286→   286→
   287→   287→  const totalPages = Math.max(1, Math.ceil(filtered.length / pageSize));
   288→   288→  const pageRows = filtered.slice(page * pageSize, page * pageSize + pageSize);
   289→   289→
   290→   290→  return (
   291→   291→    <div className="space-y-3">
   292→   292→      {(toolbar || filters || searchOf) && (
   293→   293→        <div className="flex flex-col gap-2 lg:flex-row lg:items-center lg:justify-between">
   294→   294→          <div className="flex flex-wrap items-center gap-2">{filters}</div>
   295→   295→          <div className="flex flex-wrap items-center gap-2">
   296→   296→            {searchOf ? <SearchInput value={q} onChange={setQ} placeholder={searchPlaceholder} className="w-full sm:w-64" /> : null}
   297→   297→            {toolbar}
   298→   298→          </div>
   299→   299→        </div>
   300→   300→      )}
   301→   301→      <div className="overflow-hidden rounded-lg border">
   302→   302→        <div className="max-h-[640px] overflow-auto">
   303→   303→          <Table>
   304→   304→            <TableHeader className="sticky top-0 z-10 bg-slate-50">
   305→   305→              <TableRow className="hover:bg-transparent">
   306→   306→                {columns.map((c) => (
   307→   307→                  <TableHead key={c.key} className={cn("h-9 text-xs font-semibold text-slate-600", c.headClassName)}>
   308→   308→                    <span className="inline-flex items-center gap-1">
   309→   309→                      {c.header}
   310→   310→                      {c.value ? <ChevronsUpDown className="h-3 w-3 text-slate-300" /> : null}
   311→   311→                    </span>
   312→   312→                  </TableHead>
   313→   313→                ))}
   314→   314→              </TableRow>
   315→   315→            </TableHeader>
   316→   316→            <TableBody>
   317→   317→              {pageRows.length === 0 ? (
   318→   318→                <TableRow>
   319→   319→                  <TableCell colSpan={columns.length} className="h-32 text-center text-sm text-muted-foreground">
   320→   320→                    No records found.
   321→   321→                  </TableCell>
   322→   322→                </TableRow>
   323→   323→              ) : (
   324→   324→                pageRows.map((row, i) => (
   325→   325→                  <TableRow
   326→   326→                    key={i}
   327→   327→                    className={cn(onRowClick && "cursor-pointer")}
   328→   328→                    onClick={onRowClick ? () => onRowClick(row) : undefined}
   329→   329→                  >
   330→   330→                    {columns.map((c) => (
   331→   331→                      <TableCell key={c.key} className={cn(!dense && "py-2.5", c.className)}>
   332→   332→                        {c.render ? c.render(row) : c.value ? String(c.value(row)) : null}
   333→   333→                      </TableCell>
   334→   334→                    ))}
   335→   335→                  </TableRow>
   336→   336→                ))
   337→   337→              )}
   338→   338→            </TableBody>
   339→   339→          </Table>
   340→   340→        </div>
   341→   341→      </div>
   342→   342→      <div className="flex items-center justify-between text-xs text-muted-foreground">
   343→   343→        <span>
   344→   344→          Showing {filtered.length === 0 ? 0 : page * pageSize + 1}–{Math.min(filtered.length, (page + 1) * pageSize)} of {filtered.length}
   345→   345→        </span>
   346→   346→        <div className="flex items-center gap-1">
   347→   347→          <Button variant="outline" size="sm" className="h-7 px-2" disabled={page === 0} onClick={() => setPage(page - 1)}>
   348→   348→            <ChevronLeft className="h-3.5 w-3.5" />
   349→   349→          </Button>
   350→   350→          <span className="px-1 tabular-nums">Page {page + 1} / {totalPages}</span>
   351→   351→          <Button variant="outline" size="sm" className="h-7 px-2" disabled={page >= totalPages - 1} onClick={() => setPage(page + 1)}>
   352→   352→            <ChevronRight className="h-3.5 w-3.5" />
   353→   353→          </Button>
   354→   354→        </div>
   355→   355→      </div>
   356→   356→    </div>
   357→   357→  );
   358→   358→}

   564→   564→export function Donut({
   565→   565→  data, size = 150, thickness = 22, centerLabel, centerValue,
   566→   566→}: {
   567→   567→  data: { label: string; value: number }[]; size?: number; thickness?: number;
   568→   568→  centerLabel?: string; centerValue?: string;
   569→   569→}) {
   570→   570→  const total = data.reduce((a, b) => a + b.value, 0) || 1;
   571→   571→  const r = (size - thickness) / 2;
   572→   572→  const c = 2 * Math.PI * r;
   573→   573→  const segments = React.useMemo(
   574→   574→    () =>
   575→   575→      data.map((d, i) => ({
   576→   576→        label: d.label,
   577→   577→        len: (d.value / total) * c,
   578→   578→        offset: data.slice(0, i).reduce((a, prev) => a + (prev.value / total) * c, 0),
   579→   579→        color: CHART_COLORS[i % CHART_COLORS.length],
   580→   580→      })),
   581→   581→    [data, total, c],
   582→   582→  );
   583→   583→  return (
   584→   584→    <div className="flex flex-wrap items-center gap-6">
   585→   585→      <div className="relative" style={{ width: size, height: size }}>
   586→   586→        <svg width={size} height={size} className="-rotate-90">
   587→   587→          {segments.map((s) => (
   588→   588→            <circle
   589→   589→              key={s.label} cx={size / 2} cy={size / 2} r={r} fill="none"
   590→   590→              stroke={s.color} strokeWidth={thickness}
   591→   591→              strokeDasharray={`${s.len} ${c - s.len}`} strokeDashoffset={-s.offset}
   592→   592→            />
   593→   593→          ))}
   594→   594→        </svg>
   595→   595→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   596→   596→          <span className="text-lg font-semibold text-slate-900">{centerValue}</span>
   597→   597→          <span className="text-[10px] text-muted-foreground">{centerLabel}</span>
   598→   598→        </div>
   599→   599→      </div>
   600→   600→      <ul className="space-y-1.5">
   601→   601→        {data.map((d, i) => (
   602→   602→          <li key={d.label} className="flex items-center gap-2 text-xs">
   603→   603→            <span className="h-2.5 w-2.5 rounded-sm" style={{ backgroundColor: CHART_COLORS[i % CHART_COLORS.length] }} />
   604→   604→            <span className="text-slate-600">{d.label}</span>
   605→   605→            <span className="font-semibold text-slate-900">{Math.round((d.value / total) * 100)}%</span>
   606→   606→          </li>
   607→   607→        ))}
   608→   608→      </ul>
   609→   609→    </div>
   610→   610→  );
   611→   611→}

   613→   613→export function TrendArea({
   614→   614→  data, height = 170, color = "#0d9488", format = (v: number) => inr(v, { compact: true }),
   615→   615→}: { data: { label: string; value: number }[]; height?: number; color?: string; format?: (v: number) => string }) {
   616→   616→  const w = 560; const h = height; const pad = 26;
   617→   617→  const max = Math.max(...data.map((d) => d.value)) * 1.1;
   618→   618→  const min = 0;
   619→   619→  const xs = data.map((_, i) => pad + (i * (w - pad * 2)) / (data.length - 1));
   620→   620→  const ys = data.map((d) => h - pad - ((d.value - min) / (max - min)) * (h - pad * 2));
   621→   621→  const line = xs.map((x, i) => `${i === 0 ? "M" : "L"}${x},${ys[i]}`).join(" ");
   622→   622→  const area = `${line} L${xs[xs.length - 1]},${h - pad} L${xs[0]},${h - pad} Z`;
   623→   623→  return (
   624→   624→    <div>
   625→   625→      <svg viewBox={`0 0 ${w} ${h}`} className="w-full" style={{ height }}>
   626→   626→        {[0.25, 0.5, 0.75, 1].map((f) => (
   627→   627→          <line key={f} x1={pad} x2={w - pad} y1={h - pad - f * (h - pad * 2)} y2={h - pad - f * (h - pad * 2)} stroke="#e2e8f0" strokeDasharray="3 4" />
   628→   628→        ))}
   629→   629→        <path d={area} fill={color} opacity="0.12" />
   630→   630→        <path d={line} fill="none" stroke={color} strokeWidth="2.5" strokeLinecap="round" />
   631→   631→        {xs.map((x, i) => <circle key={i} cx={x} cy={ys[i]} r="3.5" fill="white" stroke={color} strokeWidth="2" />)}
   632→   632→      </svg>

   634→   634→        {data.map((d) => (

   636→   636→            {d.label} · {format(d.value)}
